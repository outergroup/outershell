cmake_minimum_required(VERSION 3.24)
project(OuterShellWindows LANGUAGES C CXX)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION" OUTERSHELL_VERSION LIMIT_COUNT 1)
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(FILES_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../Files" CACHE PATH "Files repository")
set(TERMINAL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../Terminal" CACHE PATH "Terminal repository")
set(OUTERSHELL_PUBLIC_BASE_URL "https://outershell.org/outer-shell" CACHE STRING "Public Outer Shell release URL used for updates")
add_library(outer-windows-runtime STATIC Runtime.c CommandTasks.c AppProcess.c AppResources.c Icons.c ../Backend/OuterShellBuffer.c)
target_include_directories(outer-windows-runtime PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(outer-windows-runtime PUBLIC advapi32 shell32 ole32 bcrypt windowscodecs uuid taskschd oleaut32)
add_executable(OuterShell WIN32 BackgroundMain.c OuterShell.c Wsl.c CommandTasks.c)
target_compile_definitions(OuterShell PRIVATE OUTERSHELL_VERSION="${OUTERSHELL_VERSION}" OUTERSHELL_PUBLIC_BASE_URL="${OUTERSHELL_PUBLIC_BASE_URL}")
target_link_libraries(OuterShell PRIVATE outer-windows-runtime bcrypt taskschd oleaut32 uuid user32 gdi32 winhttp wintrust crypt32 version)
add_executable(FilesBackend WIN32 BackgroundMain.c "${FILES_ROOT}/Windows/FilesBackend.c" "${FILES_ROOT}/Windows/Environment.c")
target_include_directories(FilesBackend PRIVATE "${FILES_ROOT}/Helper" "${FILES_ROOT}/Backend/Environment")
target_link_libraries(FilesBackend PRIVATE outer-windows-runtime)
add_executable(TerminalBackend WIN32 BackgroundMain.c "${TERMINAL_ROOT}/Windows/TerminalBackend.c" "${TERMINAL_ROOT}/Backend/Session.c")
target_link_libraries(TerminalBackend PRIVATE outer-windows-runtime)
target_compile_definitions(TerminalBackend PRIVATE _CRT_SECURE_NO_WARNINGS)
add_library(outershell-core SHARED NativeRegistry.c ../Backend/OuterShellBuffer.c)
target_link_libraries(outershell-core PRIVATE bcrypt)
add_executable(outerctl ../Resources/outerctl.cpp)
target_link_libraries(outerctl PRIVATE advapi32 shell32 ws2_32)
foreach(target outer-windows-runtime OuterShell FilesBackend outershell-core outerctl)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /wd4505)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wno-unused-function)
    endif()
    if(CMAKE_C_COMPILER_ID MATCHES "Clang")
        target_compile_options(${target} PRIVATE -Wno-unused-function)
    endif()
endforeach()

option(OUTERSHELL_UPDATE_TESTS "Build the local signed update fixture" OFF)
if(OUTERSHELL_UPDATE_TESTS)
    add_executable(OuterShellIconTests test_icons.c)
    target_compile_definitions(OuterShellIconTests PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(OuterShellIconTests PRIVATE outer-windows-runtime)
    add_executable(OuterShellUpdateFixture test_updates.c Wsl.c CommandTasks.c)
    target_compile_definitions(OuterShellUpdateFixture PRIVATE _CRT_SECURE_NO_WARNINGS OUTERSHELL_VERSION="${OUTERSHELL_VERSION}" OUTERSHELL_PUBLIC_BASE_URL="${OUTERSHELL_PUBLIC_BASE_URL}")
    target_link_libraries(OuterShellUpdateFixture PRIVATE outer-windows-runtime bcrypt taskschd oleaut32 uuid user32 gdi32 winhttp wintrust crypt32 version)
endif()

foreach(app Plaintext Top)
    set(root "${CMAKE_CURRENT_SOURCE_DIR}/../../${app}")
    add_executable(${app}Backend WIN32 BackgroundMain.c AppHost.c "${root}/Windows/Environment.c")
    target_include_directories(${app}Backend PRIVATE "${root}/Helper" "${root}/Backend/Environment")
    target_link_libraries(${app}Backend PRIVATE outer-windows-runtime)
    string(TOUPPER "${app}" upper)
    string(TOLOWER "${app}" lower)
    target_compile_definitions(${app}Backend PRIVATE _CRT_SECURE_NO_WARNINGS APP_${upper}=1 APP_SERVICE="org.outershell.${app}" APP_WEB="${lower}-web" APP_BUNDLE="${app}Content.bundle" APP_INITIAL="{}")
endforeach()

foreach(app Firehose Profile)
    string(TOLOWER "${app}" lower)
    add_executable(${app}Backend WIN32 BackgroundMain.c AppTracing.c)
    target_link_libraries(${app}Backend PRIVATE outer-windows-runtime)
    target_compile_definitions(${app}Backend PRIVATE _CRT_SECURE_NO_WARNINGS APP_NAME="${app}" APP_SERVICE="org.outershell.${app}" APP_WEB="${lower}-web" APP_BUNDLE="${app}Content.bundle" APP_HELPER="/usr/local/bin/outer-${lower}-helper")
endforeach()

foreach(target PlaintextBackend TopBackend FirehoseBackend ProfileBackend)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /wd4505)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wno-unused-function)
    endif()
    if(CMAKE_C_COMPILER_ID MATCHES "Clang")
        target_compile_options(${target} PRIVATE -Wno-unused-function)
    endif()
endforeach()
