cmake_minimum_required(VERSION 3.24)
project(OuterShellWindows LANGUAGES C)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION" OUTERSHELL_VERSION LIMIT_COUNT 1)
set(CMAKE_C_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(FILES_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../Files" CACHE PATH "Files repository")
add_library(outer-windows-runtime STATIC Runtime.c ../Backend/OuterShellBuffer.c)
target_include_directories(outer-windows-runtime PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(outer-windows-runtime PUBLIC advapi32 shell32 ole32)
add_executable(OuterShell OuterShell.c Wsl.c CommandTasks.c)
target_compile_definitions(OuterShell PRIVATE OUTERSHELL_VERSION="${OUTERSHELL_VERSION}")
target_link_libraries(OuterShell PRIVATE outer-windows-runtime bcrypt taskschd oleaut32 uuid user32 gdi32 winhttp wintrust crypt32 version)
add_executable(FilesBackend "${FILES_ROOT}/Windows/FilesBackend.c")
target_link_libraries(FilesBackend PRIVATE outer-windows-runtime)
add_library(outershell-core SHARED NativeRegistry.c ../Backend/OuterShellBuffer.c)
target_link_libraries(outershell-core PRIVATE bcrypt)
foreach(target outer-windows-runtime OuterShell FilesBackend outershell-core)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /wd4505)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wno-unused-function)
    endif()
    if(CMAKE_C_COMPILER_ID MATCHES "Clang")
        target_compile_options(${target} PRIVATE -Wno-unused-function)
    endif()
endforeach()

option(OUTERSHELL_ADMIN_TESTS "Build the disposable administrator command integration fixture" OFF)
if(OUTERSHELL_ADMIN_TESTS)
    add_executable(OuterShellAdminFixture test_admin_commands.c Wsl.c CommandTasks.c)
    target_compile_definitions(OuterShellAdminFixture PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(OuterShellAdminFixture PRIVATE outer-windows-runtime bcrypt taskschd oleaut32 uuid user32 gdi32)
endif()

option(OUTERSHELL_UPDATE_TESTS "Build the local signed update fixture" OFF)

    target_link_libraries(OuterShellUpdateFixture PRIVATE outer-windows-runtime bcrypt taskschd oleaut32 uuid user32 gdi32 winhttp wintrust crypt32 version)
