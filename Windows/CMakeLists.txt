cmake_minimum_required(VERSION 3.24)
project(OuterShellWindows LANGUAGES C CXX)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION" OUTERSHELL_VERSION LIMIT_COUNT 1)
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(OUTERSHELL_PUBLIC_BASE_URL "https://outershell.org/outer-shell" CACHE STRING "Public Outer Shell release URL used for updates")
set(OUTERSHELL_ARCHITECTURE "x64" CACHE STRING "Windows architecture of this build, which selects app downloads")
set(OUTERSHELL_WINDOWS_APP "" CACHE STRING "App to build instead of Outer Shell: Files, Plaintext, Top, Firehose, Profile or Terminal")
set(APP_ROOT "" CACHE PATH "The app's repository")
add_library(outer-windows-runtime STATIC Runtime.c CommandTasks.c AppProcess.c AppResources.c Icons.c ../Backend/OuterShellBuffer.c)
target_include_directories(outer-windows-runtime PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(outer-windows-runtime PUBLIC advapi32 shell32 ole32 bcrypt windowscodecs uuid taskschd oleaut32)
set(warning_targets outer-windows-runtime)
set(app "${OUTERSHELL_WINDOWS_APP}")
string(TOUPPER "${app}" upper)
string(TOLOWER "${app}" lower)
if(app STREQUAL "")
    add_executable(OuterShell WIN32 BackgroundMain.c OuterShell.c Wsl.c CommandTasks.c)
    target_compile_options(OuterShell PRIVATE /Zi)
    target_compile_options(outer-windows-runtime PRIVATE /Zi)
    target_link_options(OuterShell PRIVATE /DEBUG:FULL /INCREMENTAL:NO /OPT:REF /OPT:ICF)
    target_compile_definitions(OuterShell PRIVATE OUTERSHELL_VERSION="${OUTERSHELL_VERSION}" OUTERSHELL_PUBLIC_BASE_URL="${OUTERSHELL_PUBLIC_BASE_URL}" OUTERSHELL_ARCHITECTURE="${OUTERSHELL_ARCHITECTURE}")
    target_link_libraries(OuterShell PRIVATE outer-windows-runtime bcrypt taskschd oleaut32 uuid user32 gdi32 winhttp wintrust crypt32 version)
    add_library(outershell-core SHARED NativeRegistry.c ../Backend/OuterShellBuffer.c)
    target_link_libraries(outershell-core PRIVATE bcrypt)
    add_executable(outerctl ../Resources/outerctl.cpp)
    target_link_libraries(outerctl PRIVATE advapi32 shell32 ws2_32)
    list(APPEND warning_targets OuterShell outershell-core outerctl)
    option(OUTERSHELL_UPDATE_TESTS "Build the local signed update fixture" OFF)
    if(OUTERSHELL_UPDATE_TESTS)
        add_executable(OuterShellIconTests test_icons.c)
        target_compile_definitions(OuterShellIconTests PRIVATE _CRT_SECURE_NO_WARNINGS)
        target_link_libraries(OuterShellIconTests PRIVATE outer-windows-runtime)
        add_executable(OuterShellContainerHelperTest test_container_helper.c)
        target_compile_definitions(OuterShellContainerHelperTest PRIVATE _CRT_SECURE_NO_WARNINGS)
        target_link_libraries(OuterShellContainerHelperTest PRIVATE outer-windows-runtime)
        add_executable(OuterShellUpdateFixture test_updates.c Wsl.c CommandTasks.c)
        target_compile_definitions(OuterShellUpdateFixture PRIVATE _CRT_SECURE_NO_WARNINGS OUTERSHELL_VERSION="${OUTERSHELL_VERSION}" OUTERSHELL_PUBLIC_BASE_URL="${OUTERSHELL_PUBLIC_BASE_URL}" OUTERSHELL_ARCHITECTURE="${OUTERSHELL_ARCHITECTURE}")
        target_link_libraries(OuterShellUpdateFixture PRIVATE outer-windows-runtime bcrypt taskschd oleaut32 uuid user32 gdi32 winhttp wintrust crypt32 version)
    endif()
elseif(app STREQUAL "Files")
    add_executable(FilesBackend WIN32 BackgroundMain.c "${APP_ROOT}/Windows/FilesBackend.c" "${APP_ROOT}/Windows/Environment.c")
    target_include_directories(FilesBackend PRIVATE "${APP_ROOT}/Helper" "${APP_ROOT}/Backend/Environment")
    target_link_libraries(FilesBackend PRIVATE outer-windows-runtime)
    list(APPEND warning_targets FilesBackend)
elseif(app STREQUAL "Terminal")
    add_executable(TerminalBackend WIN32 BackgroundMain.c "${APP_ROOT}/Windows/TerminalBackend.c" "${APP_ROOT}/Backend/Session.c" "${APP_ROOT}/Backend/Registry.c")
    target_link_libraries(TerminalBackend PRIVATE outer-windows-runtime)
    target_compile_definitions(TerminalBackend PRIVATE _CRT_SECURE_NO_WARNINGS)
elseif(app STREQUAL "Plaintext" OR app STREQUAL "Top")
    add_executable(${app}Backend WIN32 BackgroundMain.c AppHost.c "${APP_ROOT}/Windows/Environment.c")
    target_include_directories(${app}Backend PRIVATE "${APP_ROOT}/Helper" "${APP_ROOT}/Backend/Environment")
    target_link_libraries(${app}Backend PRIVATE outer-windows-runtime)
    target_compile_definitions(${app}Backend PRIVATE APP_${upper}=1 APP_SERVICE="org.outershell.${app}" APP_WEB="web" APP_BUNDLE="${app}Content.bundle" APP_INITIAL="{}")
    if(app STREQUAL "Top")
        target_sources(TopBackend PRIVATE "${APP_ROOT}/Windows/Host.c")
        target_include_directories(TopBackend PRIVATE "${APP_ROOT}/Windows")
        target_link_libraries(TopBackend PRIVATE psapi user32)
        target_compile_options(TopBackend PRIVATE /Zi)
        target_compile_options(outer-windows-runtime PRIVATE /Zi)
        target_link_options(TopBackend PRIVATE /DEBUG:FULL /INCREMENTAL:NO /OPT:REF /OPT:ICF)
        if(OUTERSHELL_UPDATE_TESTS)
            add_executable(TopHostTests "${APP_ROOT}/Windows/HostTests.c")
            target_include_directories(TopHostTests PRIVATE "${APP_ROOT}/Windows")
            target_link_libraries(TopHostTests PRIVATE outer-windows-runtime psapi user32)
            target_compile_definitions(TopHostTests PRIVATE _CRT_SECURE_NO_WARNINGS)
            add_executable(TopHostBenchmark "${APP_ROOT}/Windows/HostBenchmark.c")
            target_include_directories(TopHostBenchmark PRIVATE "${APP_ROOT}/Windows")
            target_link_libraries(TopHostBenchmark PRIVATE outer-windows-runtime psapi user32)
            target_compile_definitions(TopHostBenchmark PRIVATE _CRT_SECURE_NO_WARNINGS)
            enable_testing()
            add_test(NAME TopHostTests COMMAND TopHostTests)
        endif()
    endif()
    list(APPEND warning_targets ${app}Backend)
elseif(app STREQUAL "Firehose" OR app STREQUAL "Profile")
    add_executable(${app}Backend WIN32 BackgroundMain.c AppTracing.c)
    target_link_libraries(${app}Backend PRIVATE outer-windows-runtime)
    target_compile_definitions(${app}Backend PRIVATE APP_NAME="${app}" APP_SERVICE="org.outershell.${app}" APP_WEB="web" APP_BUNDLE="${app}Content.bundle" APP_HELPER_NAME="outer-${lower}-helper")
    list(APPEND warning_targets ${app}Backend)
else()
    message(FATAL_ERROR "Unknown OUTERSHELL_WINDOWS_APP: ${app}")
endif()
foreach(target ${warning_targets})
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /wd4505)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wno-unused-function)
    endif()
    if(CMAKE_C_COMPILER_ID MATCHES "Clang")
        target_compile_options(${target} PRIVATE -Wno-unused-function)
    endif()
endforeach()
